import HospitalCard from '../../../../components/ui/HospitalCard.astro';
import WarningBox from '../../../../components/ui/WarningBox.astro';
import DangerBox from '../../../../components/ui/DangerBox.astro';
import GlassCard from '../../../../components/ui/GlassCard.astro';

import hospitalsRaw from '../../../../data/hospitals.json';

export const hospitals = hospitalsRaw;
export const cityOrder = ['北京', '上海', '广州', '深圳', '杭州', '南京', '武汉', '成都', '重庆', '西安', '沈阳', '香港', '台北'];
export const hospitalsByCity = cityOrder
  .map((city) => ({
    city,
    hospitals: hospitals.filter((hospital) => hospital.city === city),
  }))
  .filter(({ hospitals: cityHospitals }) => cityHospitals.length > 0);
export const cityCount = hospitalsByCity.length;

<WarningBox title="I-verify bago ka pumunta">
Ang mga listahang ito ay mga tala sa access na pinananatili ng komunidad, hindi opisyal na garantiya ng institusyon. Mabilis na nagbabago ang mga patakaran, departamento, at availability ng doktor, kaya laging kailangan ang direktang kumpirmasyon.
</WarningBox>

Kasalukuyang naglalaman ang direktoryong ito ng {hospitals.length} entry na iniulat ng komunidad sa {cityCount} na lungsod. Ang pinagbatayang datos ay nananatiling pangunahin sa Tsino sa yugtong ito, kaya maaaring lumitaw pa rin sa Tsino ang mga pangalan ng ospital at mga lokal na tala.

  <>
    <h2>{city}</h2>
    {cityHospitals.map((hospital) => <HospitalCard hospital={hospital} />)}
    <hr />
  </>
))}

</GlassCard>
