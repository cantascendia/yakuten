import HospitalCard from '../../../../components/ui/HospitalCard.astro';
import WarningBox from '../../../../components/ui/WarningBox.astro';
import DangerBox from '../../../../components/ui/DangerBox.astro';
import GlassCard from '../../../../components/ui/GlassCard.astro';

import hospitalsRaw from '../../../../data/hospitals.json';

export const hospitals = hospitalsRaw;
export const cityOrder = ['北京', '上海', '广州', '深圳', '杭州', '南京', '武汉', '成都', '重庆', '西安', '沈阳', '香港', '台北'];
export const hospitalsByCity = cityOrder
  .map((city) => ({
    city,
    hospitals: hospitals.filter((hospital) => hospital.city === city),
  }))
  .filter(({ hospitals: cityHospitals }) => cityHospitals.length > 0);
export const cityCount = hospitalsByCity.length;

<WarningBox title="Verifikasi sebelum berangkat">
Daftar ini berupa catatan akses yang dikelola komunitas, bukan jaminan resmi dari institusi. Kebijakan, departemen, dan ketersediaan dokter dapat berubah dengan cepat, sehingga konfirmasi langsung selalu diperlukan.
</WarningBox>

Direktori ini saat ini mencakup {hospitals.length} entri yang dilaporkan komunitas di {cityCount} kota. Data sumber yang mendasarinya pada tahap ini masih terutama dalam bahasa Tionghoa, sehingga nama rumah sakit dan catatan lokal mungkin masih tampil dalam bahasa Tionghoa.

---

{hospitalsByCity.map(({ city, hospitals: cityHospitals }) => (
  <>
    <h2>{city}</h2>
    {cityHospitals.map((hospital) => <HospitalCard hospital={hospital} />)}
    <hr />
  </>
))}

**Cara menggunakan halaman ini**

1. Gunakan sebagai titik awal untuk verifikasi Anda sendiri.

3. Padukan akses penyedia layanan dengan perencanaan pemeriksaan laboratorium dasar dan kesiapsiagaan darurat.

</GlassCard>
