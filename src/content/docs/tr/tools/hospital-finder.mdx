import HospitalCard from '../../../../components/ui/HospitalCard.astro';
import WarningBox from '../../../../components/ui/WarningBox.astro';
import DangerBox from '../../../../components/ui/DangerBox.astro';
import GlassCard from '../../../../components/ui/GlassCard.astro';

import hospitalsRaw from '../../../../data/hospitals.json';

export const hospitals = hospitalsRaw;
export const cityOrder = ['北京', '上海', '广州', '深圳', '杭州', '南京', '武汉', '成都', '重庆', '西安', '沈阳', '香港', '台北'];
export const hospitalsByCity = cityOrder
  .map((city) => ({
    city,
    hospitals: hospitals.filter((hospital) => hospital.city === city),
  }))
  .filter(({ hospitals: cityHospitals }) => cityHospitals.length > 0);
export const cityCount = hospitalsByCity.length;

<WarningBox title="Gitmeden önce teyit edin">
Buradaki kayıtlar topluluğun güncel tuttuğu erişim notlarıdır; kurumların verdiği resmî bir garanti değildir. Kurum politikaları, bölümler ve hekimlerin müsaitliği kısa sürede değişebilir; bu yüzden gitmeden önce bilgiyi doğrudan kurumdan teyit etmek her zaman şarttır.
</WarningBox>

Bu rehberde şu anda {cityCount} şehre yayılmış, toplulukça bildirilen {hospitals.length} kayıt yer alıyor. Arkadaki kaynak veriler bu aşamada ağırlıklı olarak Çince olduğu için, hastane adları ve yerel notlar hâlâ Çince görünebilir.

{hospitalsByCity.map(({ city, hospitals: cityHospitals }) => (
  <>
    <h2>{city}</h2>
    {cityHospitals.map((hospital) => <HospitalCard hospital={hospital} />)}
    <hr />
  </>
))}

**Bu sayfayı nasıl kullanmalı**

1. Bunu kendi araştırmanız için bir başlangıç noktası olarak görün.
2. Bölümü, hekimin müsaitliğini ve kayıt kurallarını doğrudan kurumla teyit edin.
3. Sağlık kuruluşuna erişimi, başlangıç laboratuvar planlaması ve acil durum farkındalığıyla birlikte ele alın.

</GlassCard>
