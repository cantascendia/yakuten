import HospitalCard from '../../../../components/ui/HospitalCard.astro';
import WarningBox from '../../../../components/ui/WarningBox.astro';
import DangerBox from '../../../../components/ui/DangerBox.astro';
import GlassCard from '../../../../components/ui/GlassCard.astro';

import hospitalsRaw from '../../../../data/hospitals.json';

export const hospitals = hospitalsRaw;
export const cityOrder = ['北京', '上海', '广州', '深圳', '杭州', '南京', '武汉', '成都', '重庆', '西安', '沈阳', '香港', '台北'];
export const hospitalsByCity = cityOrder
  .map((city) => ({
    city,
    hospitals: hospitals.filter((hospital) => hospital.city === city),
  }))
  .filter(({ hospitals: cityHospitals }) => cityHospitals.length > 0);
export const cityCount = hospitalsByCity.length;
export const shownCount = hospitalsByCity.reduce((n, group) => n + group.hospitals.length, 0);

<WarningBox title="پیش از مراجعه راستی‌آزمایی کنید">
این فهرست‌ها یادداشت‌های دسترسی‌اند که جامعه گردآوری و نگه‌داری می‌کند، نه تضمینی رسمی از سوی خودِ مراکز درمانی. سیاست‌ها، بخش‌ها و در دسترس بودن پزشک ممکن است به‌سرعت تغییر کند؛ به همین دلیل تأیید مستقیم همیشه ضروری است.
</WarningBox>

این فهرست در حال حاضر شامل {shownCount} مورد گزارش‌شده از سوی جامعه در {cityCount} شهر است. داده‌های منبعِ زیربنایی در این مرحله بیشتر به زبان چینی است، بنابراین ممکن است نام بیمارستان‌ها و یادداشت‌های محلی همچنان به چینی نمایش داده شوند.
