import HospitalCard from '../../../../components/ui/HospitalCard.astro';
import WarningBox from '../../../../components/ui/WarningBox.astro';
import DangerBox from '../../../../components/ui/DangerBox.astro';
import GlassCard from '../../../../components/ui/GlassCard.astro';

import hospitalsRaw from '../../../../data/hospitals.json';

export const hospitals = hospitalsRaw;
export const cityOrder = ['北京', '上海', '广州', '深圳', '杭州', '南京', '武汉', '成都', '重庆', '西安', '沈阳', '香港', '台北'];
export const hospitalsByCity = cityOrder
  .map((city) => ({
    city,
    hospitals: hospitals.filter((hospital) => hospital.city === city),
  }))
  .filter(({ hospitals: cityHospitals }) => cityHospitals.length > 0);
export const cityCount = hospitalsByCity.length;

<WarningBox title="تأكّد قبل أن تذهب">
هذه القوائم ملاحظات عن الوصول إلى الرعاية يتعهّدها المجتمع بالتحديث، وليست ضمانات رسمية من المؤسسات. فالسياسات والأقسام وتوافر الأطباء قد تتغيّر بسرعة، ولهذا يبقى التأكّد المباشر ضرورياً في كل مرة.
</WarningBox>

يضمّ هذا الدليل حالياً {hospitals.length} مدخلاً وردت من المجتمع، موزّعة على {cityCount} مدينة. ولا تزال البيانات المصدرية في هذه المرحلة باللغة الصينية بالدرجة الأولى، لذا قد تظهر أسماء المستشفيات والملاحظات المحلية بالصينية.

{hospitalsByCity.map(({ city, hospitals: cityHospitals }) => (
  <>
    <h2>{city}</h2>
    {cityHospitals.map((hospital) => <HospitalCard hospital={hospital} />)}
    <hr />
  </>
))}

<GlassCard>

**كيف تستفيد من هذه الصفحة**

1. اعتبرها نقطة انطلاق لتحقّقك الخاص، لا أكثر.
2. أكّد القسم وتوافر الطبيب وقواعد التسجيل مباشرةً مع المؤسسة.
3. اربط الوصول إلى مقدّم الرعاية بالتخطيط للفحوص المخبرية الأساسية والانتباه لحالات الطوارئ.

</GlassCard>

<DangerBox title="ملاحظة بشأن الطوارئ">
هذا الدليل ليس خدمة إحالة طارئة. إن كانت لديك أعراض عاجلة، فتوجّه إلى رعاية الطوارئ أولاً بدلاً من الاعتماد على تصفّح الدليل.
</DangerBox>
