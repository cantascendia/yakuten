import HospitalCard from '../../../../components/ui/HospitalCard.astro';
import WarningBox from '../../../../components/ui/WarningBox.astro';
import DangerBox from '../../../../components/ui/DangerBox.astro';
import GlassCard from '../../../../components/ui/GlassCard.astro';

import hospitalsRaw from '../../../../data/hospitals.json';

export const hospitals = hospitalsRaw;
export const cityOrder = ['北京', '上海', '广州', '深圳', '杭州', '南京', '武汉', '成都', '重庆', '西安', '沈阳', '香港', '台北'];
export const hospitalsByCity = cityOrder
  .map((city) => ({
    city,
    hospitals: hospitals.filter((hospital) => hospital.city === city),
  }))
  .filter(({ hospitals: cityHospitals }) => cityHospitals.length > 0);
export const cityCount = hospitalsByCity.length;

<WarningBox title="ตรวจสอบก่อนเดินทางไป">
รายการเหล่านี้เป็นบันทึกการเข้าถึงบริการที่ชุมชนช่วยกันดูแล ไม่ใช่การรับรองอย่างเป็นทางการจากสถาบัน นโยบาย แผนกที่ให้บริการ และความพร้อมของแพทย์อาจเปลี่ยนแปลงได้อย่างรวดเร็ว จึงจำเป็นต้องยืนยันกับสถานพยาบาลโดยตรงเสมอ
</WarningBox>

ขณะนี้ทำเนียบนี้รวบรวมรายการที่ชุมชนช่วยกันรายงานไว้ {hospitals.length} รายการ ครอบคลุม {cityCount} เมือง ข้อมูลต้นทางในชั้นนี้ยังคงเป็นภาษาจีนเป็นหลัก ดังนั้นชื่อสถานพยาบาลและบันทึกเฉพาะพื้นที่อาจยังปรากฏเป็นภาษาจีนอยู่

    <h2>{city}</h2>

    <hr />
  </>
))}

**วิธีใช้หน้านี้**

1. ใช้เป็นจุดเริ่มต้นสำหรับการตรวจสอบยืนยันด้วยตัวคุณเอง
2. ยืนยันแผนกที่ให้บริการ ความพร้อมของแพทย์ และกฎเกณฑ์การลงทะเบียนกับสถานพยาบาลโดยตรง
3. ควบคู่ไปกับการเข้าถึงผู้ให้บริการ ให้วางแผนตรวจค่าพื้นฐาน (baseline lab) และเตรียมความพร้อมรับมือกรณีฉุกเฉินด้วย

</GlassCard>

</DangerBox>
