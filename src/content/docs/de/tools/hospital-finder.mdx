import HospitalCard from '../../../../components/ui/HospitalCard.astro';
import WarningBox from '../../../../components/ui/WarningBox.astro';
import DangerBox from '../../../../components/ui/DangerBox.astro';
import GlassCard from '../../../../components/ui/GlassCard.astro';

import hospitalsRaw from '../../../../data/hospitals.json';

export const hospitals = hospitalsRaw;
export const cityOrder = ['北京', '上海', '广州', '深圳', '杭州', '南京', '武汉', '成都', '重庆', '西安', '沈阳', '香港', '台北'];
export const hospitalsByCity = cityOrder
  .map((city) => ({
    city,
    hospitals: hospitals.filter((hospital) => hospital.city === city),
  }))
  .filter(({ hospitals: cityHospitals }) => cityHospitals.length > 0);
export const cityCount = hospitalsByCity.length;

<WarningBox title="Vor dem Besuch überprüfen">
Bei diesen Einträgen handelt es sich um von der Community gepflegte Hinweise zum Versorgungszugang, nicht um offizielle Zusicherungen der Einrichtungen. Richtlinien, Abteilungen und die Verfügbarkeit von Ärztinnen und Ärzten können sich schnell ändern, daher ist eine direkte Bestätigung immer erforderlich.
</WarningBox>

Dieses Verzeichnis umfasst derzeit {hospitals.length} von der Community gemeldete Einträge in {cityCount} Städten. Die zugrunde liegenden Quelldaten liegen in diesem Stadium noch überwiegend auf Chinesisch vor, daher können Klinikbezeichnungen und lokale Hinweise weiterhin auf Chinesisch erscheinen.
