import HospitalCard from '../../../../components/ui/HospitalCard.astro';
import WarningBox from '../../../../components/ui/WarningBox.astro';
import DangerBox from '../../../../components/ui/DangerBox.astro';
import GlassCard from '../../../../components/ui/GlassCard.astro';

import hospitalsRaw from '../../../../data/hospitals.json';

export const hospitals = hospitalsRaw;
export const cityOrder = ['北京', '上海', '广州', '深圳', '杭州', '南京', '武汉', '成都', '重庆', '西安', '沈阳', '香港', '台北'];
export const hospitalsByCity = cityOrder
  .map((city) => ({
    city,
    hospitals: hospitals.filter((hospital) => hospital.city === city),
  }))
  .filter(({ hospitals: cityHospitals }) => cityHospitals.length > 0);
export const cityCount = hospitalsByCity.length;

<WarningBox title="जाने से पहले पुष्टि कर लें">
ये सूचियाँ समुदाय द्वारा संभाले जाने वाले पहुँच-संबंधी नोट हैं — किसी संस्थान की ओर से दी गई आधिकारिक गारंटी नहीं। नीतियाँ, विभाग और डॉक्टरों की उपलब्धता बहुत जल्दी बदल सकती है, इसलिए सीधे पुष्टि कर लेना हमेशा ज़रूरी है।
</WarningBox>

इस निर्देशिका में अभी {cityCount} शहरों में फैली {hospitals.length} समुदाय-रिपोर्ट प्रविष्टियाँ शामिल हैं। मूल स्रोत डेटा इस चरण में मुख्यतः चीनी भाषा में ही है, इसलिए अस्पतालों के नाम और स्थानीय नोट अभी भी चीनी में दिख सकते हैं।

  <>
    <h2>{city}</h2>
    {cityHospitals.map((hospital) => <HospitalCard hospital={hospital} />)}
    <hr />
  </>
))}

2. विभाग, डॉक्टर की उपलब्धता और पंजीकरण के नियम सीधे संस्थान से पुष्टि कर लें।
3. देखभाल तक पहुँच के साथ-साथ आधारभूत लैब-योजना और आपातकालीन जागरूकता को भी जोड़े रखें।

</GlassCard>
